```python
from PySide6.QtCore import Slot
from views.ui_components.monitor_panels import ExplosionMonitorPanel

class MyExplosionMonitor(ExplosionMonitorPanel):
    def connect_controller_signals(self, controller):
        """连接控制器信号"""
        if controller is None:
            return
        
        # 温度数据订阅
        controller.temp_controller_updated.connect(self._on_temp_updated)
        
        # 压力数据订阅
        controller.pressure_updated.connect(self._on_pressure_updated)
        
        # 继电器状态订阅
        controller.relay_status_updated.connect(self._on_relay_updated)
        
        # 实验状态订阅
        controller.state_changed.connect(self._on_state_changed)
        
        # 进度订阅
        controller.progress_updated.connect(self._on_progress_updated)
        
        # 日志订阅
        controller.log_message.connect(self._on_log_message)
    
    @Slot(dict)
    def _on_temp_updated(self, temp_data: dict):
        """温度更新槽函数"""
        pv = temp_data.get('pv', 0.0)
        sv = temp_data.get('sv', 0.0)
        mv = temp_data.get('mv', 0.0)
        status = temp_data.get('status', '离线')
        
        # 更新显示
        self.temp_label.setText(f"{pv:.1f}°C")
        print(f"当前温度: {pv}°C, 设定: {sv}°C, 输出: {mv}%")
    
    @Slot(float)
    def _on_pressure_updated(self, pressure: float):
        """压力更新槽函数"""
        self.pressure_label.setText(f"{pressure:.2f} kPa")
        print(f"当前压力: {pressure} kPa")
    
    @Slot(dict)
    def _on_relay_updated(self, relay_states: dict):
        """继电器状态更新槽函数"""
        for relay_id, state in relay_states.items():
            print(f"继电器 {relay_id}: {'ON' if state else 'OFF'}")
    
    @Slot(str)
    def _on_state_changed(self, state: str):
        """实验状态变化槽函数"""
        state_map = {
            "idle": "⏸️ 空闲",
            "heating": "🔥 加热中",
            "spraying": "💨 喷吹中",
            "recording": "📹 录制中",
            "cooling": "❄️ 冷却中",
            "completed": "✅ 已完成"
        }
        self.status_label.setText(state_map.get(state, state))
    
    @Slot(int, str)
    def _on_progress_updated(self, progress: int, message: str):
        """进度更新槽函数"""
        self.progress_bar.setValue(progress)
        self.progress_label.setText(message)
    
    @Slot(str)
    def _on_log_message(self, message: str):
        """日志消息槽函数"""
        self.log_widget.append(message)
```

### 着火点实验监控

```python
from PySide6.QtCore import Slot
from views.ui_components.monitor_panels import IgnitionMonitorPanel

class MyIgnitionMonitor(IgnitionMonitorPanel):
    def connect_controller_signals(self, controller):
        """连接控制器信号"""
        if controller is None:
            return
        
        # 温控器数据订阅
        controller.control_temp_controller_updated.connect(self._on_furnace_temp_updated)
        
        # 样品温度订阅
        controller.sample_temps_updated.connect(self._on_sample_temps_updated)
        
        # 实验状态订阅
        controller.state_changed.connect(self._on_state_changed)
        
        # 着火检测订阅
        controller.ignition_detected.connect(self._on_ignition_detected)
        
        # 日志订阅
        controller.log_message.connect(self._on_log_message)
    
    @Slot(dict)
    def _on_furnace_temp_updated(self, temp_data: dict):
        """炉温更新槽函数"""
        pv = temp_data.get('pv', 0.0)
        self.furnace_temp_label.setText(f"{pv:.0f}°C")
    
    @Slot(list)
    def _on_sample_temps_updated(self, sample_temps: list):
        """样品温度更新槽函数"""
        for i, temp in enumerate(sample_temps):
            if i < len(self.sample_labels):
                self.sample_labels[i].setText(f"{temp:.0f}°C")
    
    @Slot(float, list)
    def _on_ignition_detected(self, threshold: float, temps: list):
        """着火检测槽函数"""
        for i, temp in enumerate(temps):
            if temp >= threshold:
                self.sample_items[i].set_ignited(True)
                print(f"🔥 样品 {i+1} 着火！温度: {temp:.1f}°C")
```

## 6. 在主窗口中集成副屏

### 完整集成示例

```python
from PySide6.QtWidgets import QMainWindow
from PySide6.QtCore import QTimer
from views.secondary_display_window import SecondaryDisplayWindow

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        
        # 副屏窗口引用
        self.secondary_display_window = None
        
        # 延迟打开副屏
        QTimer.singleShot(1000, self._open_secondary_display)
    
    def _open_secondary_display(self):
        """打开副屏窗口"""
        # 如果已存在且可见，则激活
        if self.secondary_display_window is not None:
            if self.secondary_display_window.isVisible():
                self.secondary_display_window.activateWindow()
                return
        
        try:
            # 获取控制器
            explosion_ctrl = self._get_explosion_controller()
            ignition_ctrl = self._get_ignition_controller()
            
            # 创建副屏窗口
            self.secondary_display_window = SecondaryDisplayWindow(
                explosion_controller=explosion_ctrl,
                ignition_controller=ignition_ctrl
            )
            
            # 连接关闭信号
            self.secondary_display_window.window_closed.connect(
                self._on_secondary_display_closed
            )
            
            # 显示窗口
            self.secondary_display_window.show()
            
        except Exception as e:
            print(f"打开副屏失败: {e}")
    
    def _on_secondary_display_closed(self):
        """副屏关闭回调"""
        self.secondary_display_window = None
    
    def _get_explosion_controller(self):
        """从页面获取控制器"""
        page = self.stacked_widget.widget(1)  # 爆炸性实验页面
        return getattr(page, 'controller', None)
    
    def _get_ignition_controller(self):
        """从页面获取控制器"""
        page = self.stacked_widget.widget(2)  # 着火点实验页面
        return getattr(page, 'controller', None)
```

## 7. 错误处理示例

### 安全的信号连接

```python
def connect_controller_signals(self, controller):
    """安全地连接信号"""
    if controller is None:
        print("⚠ 控制器为空，跳过信号连接")
        return
    
    try:
        # 检查信号是否存在
        if hasattr(controller, 'temp_controller_updated'):
            controller.temp_controller_updated.connect(self._on_temp_updated)
        else:
            print("⚠ 控制器缺少 temp_controller_updated 信号")
        
        # 其他信号连接...
        
        print("✅ 控制器信号连接成功")
    
    except Exception as e:
        print(f"❌ 连接控制器信号失败: {e}")
```

### 安全的信号断开

```python
def disconnect_controller_signals(self, controller):
    """安全地断开信号"""
    if controller is None:

    try:
        controller.temp_controller_updated.disconnect(self._on_temp_updated)
```

### 数据缓存优化

```python
def _poll_device_data(self):
    """轮询设备数据"""
    if not self.manager:
        return
    
    try:
        # 批量获取数据（减少查询次数）
        temp_data = self.manager.get_latest_data('温控仪表')
        pressure_data = self.manager.get_latest_data('压力仪表')
        relay_data = self.manager.get_latest_data('继电器')
        
        # 一次性发送多个信号
        if temp_data:
            self.temp_controller_updated.emit(temp_data)
        if pressure_data:
            self.pressure_updated.emit(pressure_data['pressure'])
        if relay_data:
            self.relay_status_updated.emit(relay_data['relays'])
    
    except Exception as e:
        self.logger.error(f"轮询数据失败: {e}")
```
