# 放热峰切线法实现建议

## 1. 方法原理

### 1.1 理论基础

根据 GB/T 18511-2017《煤的着火温度测定方法》，放热峰切线法是测定煤粉着火点温度的标准方法。

**原理**：
- 煤粉在升温过程中，着火时会发生明显的放热反应
- 在温度-时间曲线上表现为温度快速上升，形成"放热峰"
- 通过在放热峰的起始段和峰顶段分别作切线
- 两条切线的交点对应的温度即为着火点温度

**物理意义**：
- 起始段切线：反映正常升温阶段（外部加热为主）
- 峰顶段切线：反映放热反应阶段（内部放热为主）
- 交点：标志着火反应的开始

### 1.2 数学模型

```
设温度曲线为 T(t)，其中 T 为温度，t 为时间

1. 计算温度变化率（一阶导数）：
   dT/dt = (T[i+1] - T[i]) / (t[i+1] - t[i])

2. 计算温度变化加速度（二阶导数）：
   d²T/dt² = (dT/dt[i+1] - dT/dt[i]) / (t[i+1] - t[i])

3. 识别放热峰：
   - 寻找 d²T/dt² 显著增大的点（拐点）
   - 确定峰顶位置（dT/dt 最大值）

4. 拟合切线：
   - 起始段：在拐点前选择数据点，线性拟合得到 L1: T = k1*t + b1
   - 峰顶段：在峰顶附近选择数据点，线性拟合得到 L2: T = k2*t + b2

5. 计算交点：
   k1*t + b1 = k2*t + b2
   t_ignition = (b2 - b1) / (k1 - k2)
   T_ignition = k1*t_ignition + b1
```

## 2. 算法实现

### 2.1 核心算法伪代码

```python
def tangent_method_ignition_detection(time_data, temp_data):
    """
    放热峰切线法检测着火点
    
    参数:
        time_data: 时间序列数组
        temp_data: 温度序列数组
    
    返回:
        ignition_temp: 着火点温度
        ignition_time: 着火点时刻
    """
    
    # 1. 数据预处理：平滑滤波（减少噪声影响）
    temp_smooth = smooth_data(temp_data, window_size=5)
    
    # 2. 计算一阶导数（温度变化率）
    dT_dt = calculate_derivative(time_data, temp_smooth)
    
    # 3. 计算二阶导数（温度变化加速度）
    d2T_dt2 = calculate_derivative(time_data, dT_dt)
    
    # 4. 识别放热峰特征点
    inflection_point = find_inflection_point(d2T_dt2)  # 拐点
    peak_point = find_peak_point(dT_dt)  # 峰顶
    
    # 5. 确定拟合区间
    baseline_range = range(inflection_point - 10, inflection_point)  # 基线段
    peak_range = range(peak_point - 5, peak_point + 5)  # 峰顶段
    
    # 6. 线性拟合
    k1, b1 = linear_fit(time_data[baseline_range], temp_smooth[baseline_range])
    k2, b2 = linear_fit(time_data[peak_range], temp_smooth[peak_range])
    
    # 7. 计算交点
    t_ignition = (b2 - b1) / (k1 - k2)
    T_ignition = k1 * t_ignition + b1
    
    return T_ignition, t_ignition
```

### 2.2 Python 实现代码

```python
import numpy as np
from scipy.signal import savgol_filter
from scipy.stats import linregress

class TangentMethodDetector:
    """放热峰切线法着火点检测器"""
    
    def __init__(self, config=None):
        self.config = config or {
            'smooth_window': 11,  # 平滑窗口大小（必须为奇数）
            'smooth_order': 2,    # 平滑多项式阶数
            'baseline_points': 20,  # 基线拟合点数
            'peak_points': 10,      # 峰顶拟合点数
            'min_temp_rise': 20.0,  # 最小温升（°C）
            'min_rate_increase': 2.0  # 最小速率增幅（倍数）
        }
    
    def detect_ignition(self, time_data, temp_data):
        """
        检测着火点
        
        参数:
            time_data: numpy数组，时间数据（秒）
            temp_data: numpy数组，温度数据（°C）
        
        返回:
            result: dict {
                'ignition_temp': 着火点温度（°C）,
                'ignition_time': 着火点时刻（秒）,
                'baseline_fit': 基线拟合参数 (k1, b1),
                'peak_fit': 峰顶拟合参数 (k2, b2),
                'confidence': 置信度 (0-1)
            }
        """
        # 1. 数据验证
        if len(time_data) < 50:
            return None  # 数据点太少
        
        # 2. 数据平滑
        temp_smooth = savgol_filter(
            temp_data, 
            window_length=self.config['smooth_window'],
            polyorder=self.config['smooth_order']
        )
        
        # 3. 计算一阶导数（温升速率）
        dt = np.diff(time_data)
        dT = np.diff(temp_smooth)
        rate = dT / dt
        rate_time = time_data[:-1]  # 对应时间点
        
        # 4. 计算二阶导数（加速度）
        dt2 = np.diff(rate_time)
        drate = np.diff(rate)
        acceleration = drate / dt2
        accel_time = rate_time[:-1]
        
        # 5. 识别拐点（加速度显著增大的点）
        inflection_idx = self._find_inflection_point(
            accel_time, acceleration, rate_time, rate
        )
        
        if inflection_idx is None:
            return None  # 未找到明显的拐点
        
        # 6. 识别峰顶（温升速率最大值）
        peak_idx = self._find_peak_point(
            rate_time, rate, inflection_idx
        )
        
        if peak_idx is None:
            return None  # 未找到峰顶
        
        # 7. 拟合基线（拐点前的线性段）
        baseline_start = max(0, inflection_idx - self.config['baseline_points'])
        baseline_end = inflection_idx
        
        if baseline_end - baseline_start < 10:
            return None  # 基线点数不足
        
        k1, b1, r1, _, _ = linregress(
            time_data[baseline_start:baseline_end],
            temp_smooth[baseline_start:baseline_end]
        )
        
        # 8. 拟合峰顶线（峰顶附近的线性段）
        peak_half = self.config['peak_points'] // 2
        peak_start = max(inflection_idx, peak_idx - peak_half)
        peak_end = min(len(temp_smooth), peak_idx + peak_half)
        
        if peak_end - peak_start < 5:
            return None  # 峰顶点数不足
        
        k2, b2, r2, _, _ = linregress(
            time_data[peak_start:peak_end],
            temp_smooth[peak_start:peak_end]
        )
        
        # 9. 检查拟合质量
        if abs(r1) < 0.95 or abs(r2) < 0.90:
            return None  # 拟合质量不佳
        
        # 10. 计算交点
        if abs(k1 - k2) < 0.01:
            return None  # 两条线几乎平行
        
        t_ignition = (b2 - b1) / (k1 - k2)
        T_ignition = k1 * t_ignition + b1
        
        # 11. 验证结果合理性
        if t_ignition < time_data[inflection_idx] or t_ignition > time_data[peak_idx]:
            return None  # 交点不在合理范围内
        
        # 12. 计算置信度
        confidence = self._calculate_confidence(r1, r2, k1, k2)
        
        return {
            'ignition_temp': T_ignition,
            'ignition_time': t_ignition,
            'baseline_fit': (k1, b1),
            'peak_fit': (k2, b2),
            'confidence': confidence,
            'inflection_idx': inflection_idx,
            'peak_idx': peak_idx
        }
    
    def _find_inflection_point(self, accel_time, acceleration, rate_time, rate):
        """寻找拐点（温升速率开始显著增大的点）"""
        # 计算加速度的移动平均
        window = 5
        accel_smooth = np.convolve(
            acceleration, 
            np.ones(window)/window, 
            mode='valid'
        )
        
        # 寻找加速度超过阈值的点
        threshold = np.mean(accel_smooth) + 2 * np.std(accel_smooth)
        candidates = np.where(accel_smooth > threshold)[0]
        
        if len(candidates) == 0:
            return None
        
        # 返回第一个显著增大的点
        inflection_accel_idx = candidates[0]
        
        # 转换到原始数据索引
        inflection_time = accel_time[inflection_accel_idx]
        inflection_idx = np.argmin(np.abs(rate_time - inflection_time))
        
        return inflection_idx
    
    def _find_peak_point(self, rate_time, rate, inflection_idx):
        """寻找峰顶（温升速率最大值）"""
        # 在拐点之后寻找峰顶
        search_range = rate[inflection_idx:]
        
        if len(search_range) < 5:
            return None
        
        # 找到最大温升速率的位置
        local_peak_idx = np.argmax(search_range)
        peak_idx = inflection_idx + local_peak_idx
        
        # 验证峰顶是否显著
        peak_rate = rate[peak_idx]
        baseline_rate = np.mean(rate[max(0, inflection_idx-10):inflection_idx])
        
        if peak_rate < baseline_rate * self.config['min_rate_increase']:
            return None  # 峰不够显著
        
        return peak_idx
    
    def _calculate_confidence(self, r1, r2, k1, k2):
        """计算检测结果的置信度"""
        # 基于拟合质量和斜率差异
        fit_quality = (abs(r1) + abs(r2)) / 2
        slope_diff = abs(k2 - k1) / (abs(k1) + 1e-6)
        
        # 斜率差异越大，置信度越高
        slope_score = min(1.0, slope_diff / 10.0)
        
        # 综合得分
        confidence = 0.7 * fit_quality + 0.3 * slope_score
        
        return confidence
```

## 3. 系统集成方案

### 3.1 在 `ignition_page.py` 中添加事后分析功能

```python
from utils.tangent_method_detector import TangentMethodDetector

class IgnitionExperimentPage(QWidget):
    def __init__(self, config, ui_config, controller=None):
        super().__init__()
        # ... 现有初始化代码 ...
        
        # 添加切线法检测器
        self.tangent_detector = TangentMethodDetector()
    
    def _on_stop(self):
        """停止实验时进行事后分析"""
        # ... 现有停止代码 ...
        
        # 对每个通道进行切线法分析
        self._analyze_ignition_by_tangent_method()
    
    def _analyze_ignition_by_tangent_method(self):
        """使用切线法重新分析着火点"""
        if not self.time_history or len(self.time_history) < 50:
            return
        
        time_array = np.array(list(self.time_history))
        
        for i in range(6):
            if len(self.temp_history[i]) < 50:
                continue
            
            temp_array = np.array(list(self.temp_history[i]))
            
            # 使用切线法检测
            result = self.tangent_detector.detect_ignition(time_array, temp_array)
            
            if result is not None:
                # 对比实时检测结果
                realtime_temp = self.ignition_temperatures[i]
                tangent_temp = result['ignition_temp']
                confidence = result['confidence']
                
                # 更新标签显示两种方法的结果
                self.ignition_labels[i].setText(
                    f"着火点: {tangent_temp:.1f}°C (切线法, 置信度:{confidence:.2f})\n"
                    f"实时检测: {realtime_temp:.1f}°C" if realtime_temp else ""
                )
                
                # 记录到数据库
                if self.current_session_id is not None:
                    self.db.update_ignition_detection_tangent(
                        session_id=self.current_session_id,
                        channel=i + 1,
                        tangent_temp=tangent_temp,
                        confidence=confidence
                    )
                
                # 在图表上标注两种方法的结果
                self._plot_tangent_analysis(i, result)
    
    def _plot_tangent_analysis(self, channel, result):
        """在图表上绘制切线法分析结果"""
        # 绘制基线和峰顶拟合线
        k1, b1 = result['baseline_fit']
        k2, b2 = result['peak_fit']
        
        time_array = np.array(list(self.time_history))
        
        # 基线
        baseline = k1 * time_array + b1
        self.plot_widget.plot(time_array, baseline, pen='r', style='--')
        
        # 峰顶线
        peakline = k2 * time_array + b2
        self.plot_widget.plot(time_array, peakline, pen='b', style='--')
        
        # 标注交点
        t_ig = result['ignition_time']
        T_ig = result['ignition_temp']
        self.plot_widget.plot([t_ig], [T_ig], symbol='o', symbolSize=10, 
                             symbolBrush='red')
```

```python
def update_ignition_detection_tangent(self, session_id, channel, tangent_temp, confidence):
    """更新切线法检测结果"""
    cursor = self.conn.cursor()
    cursor.execute('''
        UPDATE ignition_detections
        SET tangent_temperature = ?, confidence = ?, method = 'tangent'
        WHERE session_id = ? AND channel = ?
    ''', (tangent_temp, confidence, session_id, channel))
    self.conn.commit()
```

## 5. 优缺点对比

| 特性 | 实时检测法 | 放热峰切线法 |
|------|-----------|-------------|
| 响应速度 | 快速（200ms） | 需要事后分析 |
| 准确性 | 较高 | 很高 |
| 标准依据 | 经验阈值 | 国家标准推荐 |
| 适用场景 | 实时监控 | 最终报告 |
| 可靠性 | 依赖参数设置 | 基于物理过程 |
| 自动化程度 | 全自动 | 需要人工审核 |
