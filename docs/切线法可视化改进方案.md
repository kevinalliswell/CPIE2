# 切线法可视化改进方案

## 问题分析

当前实现在 `_plot_tangent_analysis()` 方法中，6个样品的切线都使用相同的颜色：
- 基线：红色虚线
- 峰顶线：蓝色虚线
- 交点：橙色圆圈

这导致图表上会有6条重叠的红色基线、6条重叠的蓝色峰顶线，难以区分。

## 改进方案

### 方案一：为每个样品使用不同颜色（推荐）

让切线颜色与温度曲线颜色保持一致，便于识别。

```python
def _plot_tangent_analysis(self, channel, result, time_array, temp_array):
    """在图表上绘制切线法分析结果"""
    try:
        # 定义6个样品的颜色（与温度曲线颜色一致）
        colors = ['#ff6b6b', '#4ecdc4', '#45b7d1', '#f7dc6f', '#bb8fce', '#85c1e9']
        channel_color = colors[channel]
        
        k1, b1 = result['baseline_fit']
        k2, b2 = result['peak_fit']
        baseline_range = result['baseline_range']
        peak_range = result['peak_range']
        
        # 绘制基线（使用通道颜色的虚线）
        baseline_time = time_array[baseline_range[0]:baseline_range[1]]
        baseline_temp = k1 * baseline_time + b1
        self.plot_widget.plot(
            baseline_time, baseline_temp,
            pen=pg.mkPen(color=channel_color, width=2, style=Qt.DashLine)
        )
        
        # 绘制峰顶线（使用通道颜色的点线）
        peak_time = time_array[peak_range[0]:peak_range[1]]
        peak_temp = k2 * peak_time + b2
        self.plot_widget.plot(
            peak_time, peak_temp,
            pen=pg.mkPen(color=channel_color, width=2, style=Qt.DotLine)
        )
        
        # 标注交点（使用通道颜色）
        t_ig = result['ignition_time']
        T_ig = result['ignition_temp']
        
        scatter = pg.ScatterPlotItem(
            [t_ig], [T_ig],
            symbol='star', size=15,  # 使用星形更醒目
            brush=pg.mkBrush(color=channel_color),
            pen=pg.mkPen(color=channel_color, width=2)
        )
        self.plot_widget.addItem(scatter)
        
        # 添加文本标注（使用通道颜色）
        text = pg.TextItem(
            f"样品{channel+1}\n{T_ig:.1f}°C",
            color=channel_color,
            anchor=(0.5, 1.5),
            border=pg.mkPen(color=channel_color, width=2),
            fill=pg.mkBrush(0, 0, 0, 150)
        )
        text.setPos(t_ig, T_ig)
        self.plot_widget.addItem(text)
        
    except Exception as e:
        self._thread_safe_log(f"✗ 绘制切线法结果失败: {e}")
```

### 方案二：延长切线到交点

为了更清楚地显示切线法的原理，可以将基线和峰顶线延长到交点附近。

```python
# 延伸基线到交点

    baseline_time[0], 
    result['ignition_time'] + 5,  # 延伸到交点后5秒
    50
)

self.plot_widget.plot(
    extended_baseline_time, extended_baseline_temp,
    pen=pg.mkPen(color=channel_color, width=2, style=Qt.DashLine)
)

# 延伸峰顶线到交点
extended_peak_time = np.linspace(
    result['ignition_time'] - 5,  # 从交点前5秒开始
    peak_time[-1],
    50
)
extended_peak_temp = k2 * extended_peak_time + b2

self.plot_widget.plot(
    extended_peak_time, extended_peak_temp,
    pen=pg.mkPen(color=channel_color, width=2, style=Qt.DotLine)
)
```

## 完整改进代码

```python
def _plot_tangent_analysis(self, channel, result, time_array, temp_array):
    """在图表上绘制切线法分析结果（改进版）"""
    try:
        # 定义6个样品的颜色（与温度曲线颜色一致）
        colors = ['#ff6b6b', '#4ecdc4', '#45b7d1', '#f7dc6f', '#bb8fce', '#85c1e9']
        channel_color = colors[channel]
        
        k1, b1 = result['baseline_fit']
        k2, b2 = result['peak_fit']
        baseline_range = result['baseline_range']
        peak_range = result['peak_range']
        t_ig = result['ignition_time']
        T_ig = result['ignition_temp']
        
        # 1. 延伸并绘制基线（虚线）
        baseline_time_orig = time_array[baseline_range[0]:baseline_range[1]]
        extended_baseline_time = np.linspace(
            baseline_time_orig[0], 
            t_ig + 3,  # 延伸到交点后3秒
            50
)
        extended_baseline_temp = k1 * extended_baseline_time + b1
        
        self.plot_widget.plot(
            extended_baseline_time, extended_baseline_temp,
            pen=pg.mkPen(color=channel_color, width=2, style=Qt.DashLine),
            name=f'样品{channel+1}基线'
        )
        
        # 2. 延伸并绘制峰顶线（点线）
        peak_time_orig = time_array[peak_range[0]:peak_range[1]]
        extended_peak_time = np.linspace(
            t_ig - 3,  # 从交点前3秒开始
            peak_time_orig[-1],
            50
        )
        extended_peak_temp = k2 * extended_peak_time + b2
        
        self.plot_widget.plot(
            extended_peak_time, extended_peak_temp,
            pen=pg.mkPen(color=channel_color, width=2, style=Qt.DotLine),
            name=f'样品{channel+1}峰顶线'
        )
        
        # 3. 绘制交点标记（星形）
        scatter = pg.ScatterPlotItem(
            [t_ig], [T_ig],
            symbol='star', size=15,
            brush=pg.mkBrush(color=channel_color),
            pen=pg.mkPen(color=channel_color, width=2)
        )
        self.plot_widget.addItem(scatter)
        
        # 4. 添加垂直辅助线（从交点到x轴，半透明）
        inf_line_v = pg.InfiniteLine(
            pos=t_ig,
            angle=90,
            pen=pg.mkPen(color=channel_color, width=1, style=Qt.DotLine, alpha=128)
        )
        self.plot_widget.addItem(inf_line_v)
        
        # 5. 添加文本标注
        text = pg.TextItem(
            f"样品{channel+1}\n{T_ig:.1f}°C\n置信度:{result['confidence']:.2f}",
            color=channel_color,
            anchor=(0.5, 1.5),
            border=pg.mkPen(color=channel_color, width=2),
            fill=pg.mkBrush(0, 0, 0, 180)  # 半透明黑色背景
        )
        text.setPos(t_ig, T_ig)
        self.plot_widget.addItem(text)
        
        # 6. 保存绘制项的引用，便于后续控制显示/隐藏
        if not hasattr(self, 'tangent_plot_items'):
            self.tangent_plot_items = [[] for _ in range(6)]
        
        # 将本次绘制的所有项目保存起来
        # （这里需要保存所有 plot_widget.plot() 的返回值）
        
    except Exception as e:
        self._thread_safe_log(f"✗ 绘制切线法结果失败: {e}")
```

## 其他建议

### 1. 添加图例说明

在图表中添加图例，说明不同线型的含义：
- 实线：温度曲线
- 虚线：基线拟合
- 点线：峰顶拟合
- 星形：着火点

### 2. 优化文本标注位置

如果多个样品的着火点温度接近，文本标注可能会重叠。可以：
- 自动检测重叠
- 调整标注位置
- 使用引线连接标注和点

### 3. 分组显示

可以考虑将6个样品分成两组（上下或左右），每组3个样品，避免拥挤。

## 总结

**推荐使用方案一**：为每个样品使用不同颜色，并延长切线到交点附近。这样可以：
1. 清晰区分不同样品
2. 直观展示切线法原理
3. 保持图表整洁
